-- Таблица тестов
CREATE TABLE IF NOT EXISTS tests (
    id SERIAL PRIMARY KEY,
    name VARCHAR(200) NOT NULL,
    description TEXT,
    created_by INT REFERENCES users(id) ON DELETE SET NULL,
    questions_per_session INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Параметры (шкалы)
CREATE TABLE IF NOT EXISTS parameters (
    id SERIAL PRIMARY KEY,
    test_id INT NOT NULL REFERENCES tests(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,
    scale_type VARCHAR(20) NOT NULL CHECK (scale_type IN ('BINARY','RANGE')),
    min_value INT DEFAULT 0,
    max_value INT DEFAULT 0
);

-- Интерпретации параметров
CREATE TABLE IF NOT EXISTS parameter_interpretations (
    id SERIAL PRIMARY KEY,
    parameter_id INT NOT NULL REFERENCES parameters(id) ON DELETE CASCADE,
    range_start INT,
    range_end INT,
    binary_value VARCHAR(10),
    interpretation_text TEXT NOT NULL
);

-- Вопросы (банк вопросов)
CREATE TABLE IF NOT EXISTS questions (
    id SERIAL PRIMARY KEY,
    test_id INT NOT NULL REFERENCES tests(id) ON DELETE CASCADE,
    text TEXT NOT NULL,
    order_num INT NOT NULL
);

-- Варианты ответов
CREATE TABLE IF NOT EXISTS answer_options (
    id SERIAL PRIMARY KEY,
    question_id INT NOT NULL REFERENCES questions(id) ON DELETE CASCADE,
    text VARCHAR(500) NOT NULL,
    order_num INT NOT NULL
);

-- Влияние ответа на параметры
CREATE TABLE IF NOT EXISTS answer_parameter_impact (
    id SERIAL PRIMARY KEY,
    answer_option_id INT NOT NULL REFERENCES answer_options(id) ON DELETE CASCADE,
    parameter_id INT NOT NULL REFERENCES parameters(id) ON DELETE CASCADE,
    delta INT NOT NULL
);

-- Сессии прохождения теста
CREATE TABLE IF NOT EXISTS test_sessions (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id) ON DELETE CASCADE,
    test_id INT REFERENCES tests(id) ON DELETE CASCADE,
    start_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    end_time TIMESTAMP,
    status VARCHAR(20) DEFAULT 'IN_PROGRESS'
);

-- Ответы пользователя в сессии
CREATE TABLE IF NOT EXISTS user_answers (
    id SERIAL PRIMARY KEY,
    session_id INT REFERENCES test_sessions(id) ON DELETE CASCADE,
    question_id INT REFERENCES questions(id),
    answer_option_id INT REFERENCES answer_options(id),
    answered_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Результаты сессии по каждому параметру (сырые и масштабированные)
CREATE TABLE IF NOT EXISTS test_results (
    id SERIAL PRIMARY KEY,
    session_id INT REFERENCES test_sessions(id) ON DELETE CASCADE,
    parameter_id INT REFERENCES parameters(id),
    raw_score INT,
    scaled_score INT,
    interpreted_code VARCHAR(10),
    interpretation_text TEXT
);

-- Назначения тестов (пользователю или группе)
CREATE TABLE IF NOT EXISTS test_assignments (
    id SERIAL PRIMARY KEY,
    test_id INT REFERENCES tests(id) ON DELETE CASCADE,
    assigned_by INT REFERENCES users(id),
    assigned_to_user INT REFERENCES users(id),
    assigned_to_group INT REFERENCES groups(id),
    assigned_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    due_date DATE
);

-- Индексы для скорости
CREATE INDEX IF NOT EXISTS idx_parameters_test ON parameters(test_id);
CREATE INDEX IF NOT EXISTS idx_questions_test ON questions(test_id);
CREATE INDEX IF NOT EXISTS idx_answer_options_question ON answer_options(question_id);
CREATE INDEX IF NOT EXISTS idx_impacts_answer ON answer_parameter_impact(answer_option_id);
CREATE INDEX IF NOT EXISTS idx_sessions_user ON test_sessions(user_id);
CREATE INDEX IF NOT EXISTS idx_sessions_test ON test_sessions(test_id);
